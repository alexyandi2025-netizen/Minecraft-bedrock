                                                                                                                                            const scene = new THREE.Scene();

                                                                                                                                                            scene.fog = new THREE.Fog(0x87CEEB, 5, 30);

                                                                                                                                                                    const camera = new THREE.PerspectiveCamera(75, window.innerWidth/window.innerHeight, 0.1, 1000);
                                                                                                                                                                            const renderer = new THREE.WebGLRenderer({ antialias: false });
                                                                                                                                                                                    renderer.setSize(window.innerWidth, window.innerHeight);
                                                                                                                                                                                            document.body.appendChild(renderer.domElement);

                                                                                                                                                                                                    scene.add(new THREE.AmbientLight(0xffffff, 1.2));

                                                                                                                                                                                                            const geo = new THREE.BoxGeometry(1, 1, 1);
                                                                                                                                                                                                                    const mat = new THREE.MeshLambertMaterial({ color: 0x5da341 });
                                                                                                                                                                                                                            
                                                                                                                                                                                                                                    let chunks = {}; 
                                                                                                                                                                                                                                            const renderDist = 12; 

                                                                                                                                                                                                                                                    // Función simple para generar altura (Pseudo-Perlin)
                                                                                                                                                                                                                                                            function getHeight(x, z) {
                                                                                                                                                                                                                                                                        return Math.floor(Math.sin(x/4) * Math.cos(z/4) * 2.5);
                                                                                                                                                                                                                                                                                }

                                                                                                                                                                                                                                                                                        function updateWorld() {
                                                                                                                                                                                                                                                                                                    const px = Math.floor(camera.position.x);
                                                                                                                                                                                                                                                                                                                const pz = Math.floor(camera.position.z);

                                                                                                                                                                                                                                                                                                                            for (let x = px - renderDist; x < px + renderDist; x++) {
                                                                                                                                                                                                                                                                                                                                            for (let z = pz - renderDist; z < pz + renderDist; z++) {
                                                                                                                                                                                                                                                                                                                                                                const id = `${x},${z}`;
                                                                                                                                                                                                                                                                                                                                                                                    if (!chunks[id]) {
                                                                                                                                                                                                                                                                                                                                                                                                            const h = getHeight(x, z);
                                                                                                                                                                                                                                                                                                                                                                                                                                    const b = new THREE.Mesh(geo, mat);
                                                                                                                                                                                                                                                                                                                                                                                                                                                            b.position.set(x, h, z);
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                    scene.add(b);
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                            chunks[id] = { mesh: b, y: h };
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                }
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                }
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                            }

                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                        for (let id in chunks) {
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                        const [cx, cz] = id.split(',').map(Number);
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                        if (Math.abs(cx - px) > renderDist + 2 || Math.abs(cz - pz) > renderDist + 2) {
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                            scene.remove(chunks[id].mesh);
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                delete chunks[id];
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                }
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                            }
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                    }

                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                            let velY = 0;
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                    let jumping = false;
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                            const playerHeight = 1.8;

                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                    function updatePhysics() {
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                const px = Math.round(camera.position.x);
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                            const pz = Math.round(camera.position.z);
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                        const id = `${px},${pz}`;
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                    
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                // Si hay un bloque debajo, obtenemos su altura, si no, usamos una base
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                            const currentFloorY = chunks[id] ? chunks[id].y + playerHeight : -5;

                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                        velY -= 0.012; // Gravedad
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                    camera.position.y += velY;

                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                // Colisión con el suelo dinámico
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                            if (camera.position.y <= currentFloorY) {
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                            camera.position.y = currentFloorY;
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                            velY = 0;
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                            jumping = false;
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                        }
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                }

                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                            });

                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                e.preventDefault();
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                            if (!jumping) {
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                            velY = 0.24;
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                            jumping = true;

                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                    lon -= (e.touches[0].pageX - sX) * 0.4;
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                    lat = Math.max(-85, Math.min(85, lat - (e.touches[0].pageY - sY) * 0.4)); 
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                    sX = e.touches[0].pageX; sY = e.touches[0].pageY;
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                }
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                        });

                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                function animate() {
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                            requestAnimationFrame(animate);
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                        const phi = THREE.MathUtils.degToRad(90-lat), theta = THREE.MathUtils.degToRad(lon);
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                    camera.lookAt(camera.position.clone().add(new THREE.Vector3(Math.sin(phi)*Math.cos(theta), Math.cos(phi), Math.sin(phi)*Math.sin(theta))));

                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                        const side = new THREE.Vector3().crossVectors(new THREE.Vector3(0,1,0), dir).normalize();

                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                        if(keys.s) camera.position.addScaledVector(dir, -spd);
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                    if(keys.a) camera.position.addScaledVector(side, spd);
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                if(keys.d) camera.position.addScaledVector(side, -spd);

                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                    updatePhysics();
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                renderer.render(scene, camera);
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                        }
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                camera.position.set(0, 5, 0);
